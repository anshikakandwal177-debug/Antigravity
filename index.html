RFM Customer Scoring via NTILE() Window Functions

Categorizes active shoppers into 4 quantiles based on purchase frequency, monetary value, and recent activity.

WITH CustomerAggregates AS (
  SELECT
    customer_id,
    MAX(order_date) AS last_purchase_date,
    COUNT(order_id) AS total_orders,
    SUM(order_amount) AS total_spend
  FROM orders
  WHERE status = 'COMPLETED'
  GROUP BY customer_id
),
RFMScores AS (
  SELECT
    customer_id,
    -- Recency: Higher score for more recent orders
    NTILE(4) OVER (ORDER BY last_purchase_date ASC) AS r_score,
    -- Frequency: Higher score for repeat transactions
    NTILE(4) OVER (ORDER BY total_orders ASC) AS f_score,
    -- Monetary: Higher score for greater revenue
    NTILE(4) OVER (ORDER BY total_spend ASC) AS m_score
  FROM CustomerAggregates
)
SELECT
  customer_id,
  r_score, f_score, m_score,
  CASE
    WHEN r_score = 4 AND f_score >= 3 THEN 'Champions / VIP'
    WHEN r_score <= 2 AND f_score >= 3 THEN 'At Risk / High Value'
    WHEN r_score <= 2 AND f_score <= 2 THEN 'Hibernating'
    ELSE 'Promising'
  END AS customer_segment
FROM RFMScores;

Month-over-Month Growth with Common Table Expressions

Calculates running revenue and percentage month-over-month growth rate without cartesian joins.

WITH MonthlySales AS (
  SELECT
    DATE_TRUNC('month', order_date) AS sales_month,
    SUM(net_amount) AS monthly_revenue
  FROM sales_transactions
  GROUP BY DATE_TRUNC('month', order_date)
),
SalesWithLag AS (
  SELECT
    sales_month,
    monthly_revenue,
    LAG(monthly_revenue, 1) OVER (ORDER BY sales_month) AS prev_month_revenue,
    SUM(monthly_revenue) OVER (ORDER BY sales_month ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS cumulative_ytd_revenue
  FROM MonthlySales
)
SELECT
  TO_CHAR(sales_month, 'YYYY-MM') AS month_formatted,
  monthly_revenue,
  cumulative_ytd_revenue,
  ROUND(
    ((monthly_revenue - prev_month_revenue) / NULLIF(prev_month_revenue, 0)) * 100, 2
  ) AS mom_growth_percentage
FROM SalesWithLag
ORDER BY sales_month DESC;

Third Normal Form (3NF) Relational Architecture DDL

Clean, normalized schema preventing insertion/deletion anomalies and maintaining referential integrity.

-- 1NF & 2NF Ensured: Atomic attributes and full key dependencies
CREATE TABLE product_categories (
  category_id SERIAL PRIMARY KEY,
  category_name VARCHAR(100) NOT NULL UNIQUE,
  department VARCHAR(50) NOT NULL
);

CREATE TABLE products (
  product_id SERIAL PRIMARY KEY,
  product_name VARCHAR(150) NOT NULL,
  category_id INT NOT NULL REFERENCES product_categories(category_id) ON DELETE RESTRICT,
  unit_price NUMERIC(10, 2) CHECK (unit_price > 0),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 3NF: No transitive dependency of customer details on order_id
CREATE TABLE order_items (
  order_item_id SERIAL PRIMARY KEY,
  order_id INT NOT NULL REFERENCES orders(order_id) ON DELETE CASCADE,
  product_id INT NOT NULL REFERENCES products(product_id),
  quantity INT NOT NULL CHECK (quantity > 0),
  price_at_purchase NUMERIC(10, 2) NOT NULL,
  CONSTRAINT uq_order_product UNIQUE(order_id, product_id)
);

-- Indexing for high-throughput join lookups
CREATE INDEX idx_orders_customer ON orders(customer_id);
CREATE INDEX idx_order_items_product ON order_items(product_id);

Days-Between-Visits & Inactivity Flags via LAG()

Evaluates session intervals per user to flag declining engagement before churn happens.

WITH UserActivitySeq AS (
  SELECT
    user_id,
    session_timestamp,
    LAG(session_timestamp, 1) OVER (
      PARTITION BY user_id 
      ORDER BY session_timestamp
    ) AS prev_session_time
  FROM app_sessions
)
SELECT
  user_id,
  session_timestamp,
  EXTRACT(DAY FROM (session_timestamp - prev_session_time)) AS days_since_last_visit,
  CASE
    WHEN prev_session_time IS NULL THEN 'First Visit'
    WHEN EXTRACT(DAY FROM (session_timestamp - prev_session_time)) > 30 THEN 'Re-Engaged'
    ELSE 'Active Retention'
  END AS engagement_state
FROM UserActivitySeq;
